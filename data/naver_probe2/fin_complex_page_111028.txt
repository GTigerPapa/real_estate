      var $ogUrlMeta = document.createElement("meta");

      $ogUrlMeta.setAttribute("property", "og:url");
      $ogUrlMeta.setAttribute("content", window.location.href);

      document.head.appendChild($ogUrlMeta);

      var urlParams = new URLSearchParams(window.location.search);

      var matched =
        /NAVER\([inapp|higgs]+;[^0-9]*(search|pay);[^0-9]*(\d+);[^0-9]*(\d+.\d+.\d+).*\)/im.exec(
          navigator.userAgent
        );
      var isPayApp = matched ? matched[1] === "pay" : false;
      var buttonText = urlParams.get("text");

      function isSafeUrl(inputUrl, base) {
        var allowedProtocols = [
          "naverpayapp:",
          "http:",
          "https:",
          "intent:",
          "naverapp:",
          "navercafe:",
          "naversearchapp:",
          "navershopping:",
          "bandapp:",
        ];

        if (!inputUrl || !inputUrl.trim()) {
          return false;
        }

        var urlLike;
        try {
          urlLike = new URL(inputUrl, base);
        } catch (e) {
          return false;
        }

        if (allowedProtocols.indexOf(urlLike.protocol) === -1) {
          return false;
        }
        if (urlLike.protocol === "http:" || urlLike.protocol === "https:") {
          return /(\.(naver|navercorp)\.com|\/naver\.me)$/.test(
            urlLike.origin.replace(/:\d+$/, "")
          );
        }
        return true;
      }

      if (window.history.length === 1) {
        document.getElementById("prev_button").textContent = buttonText
          ? decodeURIComponent(buttonText)
          : "홈으로 이동";
      }

      function goPayHome() {
        if (
          isPayApp &&
          window.payInAppBrowser &&
          typeof window.payInAppBrowser.showServiceMenu === "function"
        ) {
          window.payInAppBrowser.showServiceMenu?.(
            '{"menuName": "payment", "subMenuName": "pay"}'
          );
        } else {
          location.replace("https://home.pay.naver.com/");
        }
      }

      function goBack(custom) {
        if (window.history.length > 1) {
          window.history.back();
        } else {
          if (custom) {
            var url = urlParams.get("url");

            if (url) {
              var decodedUrl = decodeURIComponent(url);

              if (isSafeUrl(decodedUrl)) {
                location.replace(decodedUrl);
                return;
              }
            }
          }

          goPayHome();
        }
      }
